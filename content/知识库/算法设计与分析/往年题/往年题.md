$$
\boxed{
\begin{aligned}
&\text{\textbf{Algorithm BranchAndBound01Knapsack}} \\
&\quad \text{\textbf{Input}: } \text{物品重量 } w[1..n],\ \text{背包容量 } M \\
&\quad \text{\textbf{Output}: } \text{最大重量 } bestvalue \\
&\quad \text{\textbf{Procedure}:} \\
&\quad 1.\ \text{初始化优先队列 } Q \text{（最大堆）},\ bestvalue \leftarrow 0 \\
&\quad 2.\ \text{将物品按重量非递减排序} \\
&\quad 3.\ \text{创建根节点 } u: \\
&\quad \quad \quad u.level \leftarrow 0,\ u.s \leftarrow 0,\ u.r \leftarrow \sum_{i=1}^n w[i] \\
&\quad \quad \quad u.upperbound \leftarrow u.s + u.r \\
&\quad \quad \quad Q.\text{push}(u) \\
&\quad 4.\ \mathbf{while}\ Q \neq \emptyset: \\
&\quad \quad \quad u \leftarrow Q.\text{pop}() \\
&\quad \quad \quad \mathbf{if}\ u.upperbound \leq bestvalue: \\
&\quad \quad \quad \quad \mathbf{continue} \quad \textcolor{gray}{// 剪枝} \\
&\quad \quad \quad \mathbf{if}\ u.level == n: \quad \textcolor{gray}{// 完整解} \\
&\quad \quad \quad \quad \mathbf{if}\ u.s > bestvalue: \\
&\quad \quad \quad \quad \quad bestvalue \leftarrow u.s \\
&\quad \quad \quad \mathbf{else}: \\
&\quad \quad \quad \quad k \leftarrow u.level + 1 \\
&\quad \quad \quad \quad \text{生成左子节点 } vleft: \quad \textcolor{gray}{// 选第k件物品} \\
&\quad \quad \quad \quad \quad \mathbf{if}\ u.s + w[k] \leq M: \quad \textcolor{gray}{// 限界条件a} \\
&\quad \quad \quad \quad \quad \quad vleft.level \leftarrow k \\
&\quad \quad \quad \quad \quad \quad vleft.s \leftarrow u.s + w[k] \\
&\quad \quad \quad \quad \quad \quad vleft.r \leftarrow u.r - w[k] \\
&\quad \quad \quad \quad \quad \quad vleft.upperbound \leftarrow vleft.s + vleft.r \\
&\quad \quad \quad \quad \quad \quad \mathbf{if}\ vleft.upperbound > bestvalue: \\
&\quad \quad \quad \quad \quad \quad \quad Q.\text{push}(vleft) \\
&\quad \quad \quad \quad \text{生成右子节点 } vright: \quad \textcolor{gray}{// 不选第k件物品} \\
&\quad \quad \quad \quad \quad vright.level \leftarrow k \\
&\quad \quad \quad \quad \quad vright.s \leftarrow u.s \\
&\quad \quad \quad \quad \quad vright.r \leftarrow u.r - w[k] \\
&\quad \quad \quad \quad \quad vright.upperbound \leftarrow vright.s + vright.r \\
&\quad \quad \quad \quad \quad \mathbf{if}\ vright.upperbound > bestvalue\ \mathbf{and}\ vright.s + vright.r \geq M: \quad \textcolor{gray}{// 限界条件b} \\
&\quad \quad \quad \quad \quad \quad Q.\text{push}(vright) \\
&\quad 5.\ \mathbf{return}\ bestvalue \\
&\quad \mathbf{end\ procedure}
\end{aligned}
}
$$
