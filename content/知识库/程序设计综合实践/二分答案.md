```C++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

/**
 * 核心思路：
 * 1. 严格单调递增要求 B[i] < B[i+1]，且 B 为整数。
 *    等价于 B[i] - i < B[i+1] - (i+1)。
 *    令 C[i] = A[i] - i，问题转化为：
 *    寻找序列 D，使得 D 为非递减序列 (D[i] <= D[i+1])，
 *    且最小化 max|C[i] - D[i]|。
 *
 * 2. 最小化最大差值问题，可以考虑二分答案。
 *    假设代价为 mid，那么对于每个位置 i，B[i] 必须落在 [A[i] - mid, A[i] + mid] 范围内。
 *    同时要满足 B[i] - i <= B[i+1] - (i+1)。
 *    令 C[i] = A[i] - i，则 D[i] 必须落在 [C[i] - mid, C[i] + mid] 范围内，且 D[i] <= D[i+1]。
 */

bool check(int mid, int n, const vector<int> &C)
{
    // 设置第一个元素的最小可能值
    int current_min = C[0] - mid;

    for (int i = 1; i < n; ++i)
    {
        // 当前元素能达到的范围是 [C[i] - mid, C[i] + mid]
        // 且必须满足 D[i] >= D[i-1] (即 current_min)
        int lower = C[i] - mid;
        int upper = C[i] + mid;

        // 如果当前元素的上限都比前一个元素的最小值还要小，说明 mid 太小了
        if (upper < current_min)
            return false;

        // 为了给后面留出更多空间，我们让 D[i] 取尽可能小的值
        // 但不能小于前一个元素的值，也不能小于它的允许下限
        current_min = max(current_min, lower);
    }
    return true;
}

void solve(int caseNum)
{
    int n;
    cin >> n;
    vector<int> A(n);
    vector<int> C(n);
    for (int i = 0; i < n; ++i)
    {
        cin >> A[i];
        // 关键变换：C[i] = A[i] - i (下标从0开始)
        // 这里的 i 也就是论文中常见的转换技巧
        C[i] = A[i] - i;
    }

    // 二分搜索最小代价 mid
    int left = 0, right = 1000000; // A[i] 最大 10^6，代价不会超过这个范围
    int ans = right;

    while (left <= right)
    {
        int mid = left + (right - left) / 2;
        if (check(mid, n, C))
        {
            ans = mid;
            right = mid - 1;
        }
        else
        {
            left = mid + 1;
        }
    }

    cout << "Case #" << caseNum << ":" << endl;
    cout << ans << endl;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    if (!(cin >> T))
        return 0;
    for (int i = 1; i <= T; ++i)
    {
        solve(i);
    }
    return 0;
}
```
