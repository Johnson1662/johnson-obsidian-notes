> [!NOTE] 序列变换
>
> ### 题目描述
>
> 给定序列 $A=\{A_1, A_2, \dots, A_n\}$, 要求改变序列 $A$ 中的某些元素，形成一个**严格单调**的序列 $B$（即 $B_i < B_{i+1}, 1 \le i < N$）。
>
> 定义变换代价为 $\text{cost}(A, B) = \max(|A_i - B_i|)$。请求出满足条件的**最小代价**。
>
> > [!important] 注意
> > 每个元素在变换前后都必须是**整数**。
>
> ### 输入格式
>
> - 第一行为测试组数 $T$ ($1 \le T \le 10$)。
> - 每组数据：
>   - 第一行：序列长度 $N$ ($1 \le N \le 10^5$)。
>   - 第二行：$N$ 个正整数 $A_1, A_2, \dots, A_n$ ($A_i \le 10^6$)。
>
> ### 输出格式
>
> 对于每个样例，输出两行：
>
> 1. `Case #i:`
> 2. 最小代价整数。
>
> ### 样例
>
> **Sample Input**
>
> ```text
> 2
> 2
> 1 10
> 3
> 2 5 4
> ```
>
> **Sample Output**
>
> ```text
> Case #1:
> 0
> Case #2:
> 1
> ```

```C++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

/**
 * 核心思路：
 * 1. 严格单调递增要求 B[i] < B[i+1]，且 B 为整数。
 *    等价于 B[i] - i < B[i+1] - (i+1)。
 *    令 C[i] = A[i] - i，问题转化为：
 *    寻找序列 D，使得 D 为非递减序列 (D[i] <= D[i+1])，
 *    且最小化 max|C[i] - D[i]|。
 *
 * 2. 最小化最大差值问题，可以考虑二分答案。
 *    假设代价为 mid，那么对于每个位置 i，B[i] 必须落在 [A[i] - mid, A[i] + mid] 范围内。
 *    同时要满足 B[i] - i <= B[i+1] - (i+1)。
 *    令 C[i] = A[i] - i，则 D[i] 必须落在 [C[i] - mid, C[i] + mid] 范围内，且 D[i] <= D[i+1]。
 */

bool check(int mid, int n, const vector<int> &C)
{
    // 设置第一个元素的最小可能值
    int current_min = C[0] - mid;

    for (int i = 1; i < n; ++i)
    {
        // 当前元素能达到的范围是 [C[i] - mid, C[i] + mid]
        // 且必须满足 D[i] >= D[i-1] (即 current_min)
        int lower = C[i] - mid;
        int upper = C[i] + mid;

        // 如果当前元素的上限都比前一个元素的最小值还要小，说明 mid 太小了
        if (upper < current_min)
            return false;

        // 为了给后面留出更多空间，我们让 D[i] 取尽可能小的值
        // 但不能小于前一个元素的值，也不能小于它的允许下限
        current_min = max(current_min, lower);
    }
    return true;
}

void solve(int caseNum)
{
    int n;
    cin >> n;
    vector<int> A(n);
    vector<int> C(n);
    for (int i = 0; i < n; ++i)
    {
        cin >> A[i];
        // 关键变换：C[i] = A[i] - i (下标从0开始)
        // 这里的 i 也就是论文中常见的转换技巧
        C[i] = A[i] - i;
    }

    // 二分搜索最小代价 mid
    int left = 0, right = 1000000; // A[i] 最大 10^6，代价不会超过这个范围
    int ans = right;

    while (left <= right)
    {
        int mid = left + (right - left) / 2;
        if (check(mid, n, C))
        {
            ans = mid;
            right = mid - 1;
        }
        else
        {
            left = mid + 1;
        }
    }

    cout << "Case #" << caseNum << ":" << endl;
    cout << ans << endl;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int T;
    if (!(cin >> T))
        return 0;
    for (int i = 1; i <= T; ++i)
    {
        solve(i);
    }
    return 0;
}
```

River Hopscotch

Contest Problem

Time Limit：C/C++ 2000MS，Other 4000MS  
Memory Limit：C/C++ 64MB，Other 128MB

Description

Every year the cows hold an event featuring a peculiar version of hopscotch that involves carefully jumping from rock to rock in a river. The excitement takes place on a long, straight river with a rock at the start and another rock at the end, *L* units away from the start (1 ≤ *L* ≤ 1,000,000,000). Along the river between the starting and ending rocks, *N* (0 ≤ *N* ≤ 50,000) more rocks appear, each at an integral distance *Di* from the start (0 < *Di* < *L*).

To play the game, each cow in turn starts at the starting rock and tries to reach the finish at the ending rock, jumping only from rock to rock. Of course, less agile cows never make it to the final rock, ending up instead in the river.

Farmer John is proud of his cows and watches this event each year. But as time goes by, he tires of watching the timid cows of the other farmers limp across the short distances between rocks placed too closely together. He plans to remove several rocks in order to increase the shortest distance a cow will have to jump to reach the end. He knows he cannot remove the starting and ending rocks, but he calculates that he has enough resources to remove up to *M* rocks (0 ≤ *M* ≤ *N*).

FJ wants to know exactly how much he can increase the shortest distance **before** he starts removing the rocks. Help Farmer John determine the greatest possible shortest distance a cow has to jump after removing the optimal set of *M* rocks.

Input

Line 1: Three space-separated integers: *L*, *N*, and *M*  
Lines 2…_N_+1: Each line contains a single integer indicating how far some rock is away from the starting rock. No two rocks share the same position.

Output

Line 1: A single integer that is the maximum of the shortest distance a cow has to jump after removing *M* rocks

Sample Input 1 

25 5 2
2
14
11
21
17

Sample Output 1 

4

Hint

Before removing any rocks, the shortest jump was a jump of 2 from 0 (the start) to 2. After removing the rocks at 2 and 14, the shortest required jump is a jump of 4 (from 17 to 21 or from 21 to 25).

```C++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

bool check(long long min_dist, int L, int N, int M, const vector<int> &rocks)
{
    int removed = 0;
    int last_pos = 0;

    for (int i = 0; i < N; ++i)
    {
        if (rocks[i] - last_pos < min_dist)
        {
            // Dist too short, remove this rock
            removed++;
        }
        else
        {
            // Keep this rock, jump to it
            last_pos = rocks[i];
        }
    }

    // Check the last jump to the exit
    if (L - last_pos < min_dist)
    {
        removed++;
    }

    return removed <= M;
}

int main()
{
    int L, N, M;
    if (!(cin >> L >> N >> M))
        return 0;

    vector<int> rocks(N);
    for (int i = 0; i < N; ++i)
    {
        cin >> rocks[i];
    }

    sort(rocks.begin(), rocks.end());

    long long left = 0, right = L;
    long long ans = 0;

    while (left <= right)
    {
        long long mid = left + (right - left) / 2;
        if (mid == 0)
        { // Avoid infinite loop or logic errors for 0 distance
            left = 1;
            continue;
        }
        if (check(mid, L, N, M, rocks))
        {
            ans = mid;
            left = mid + 1;
        }
        else
        {
            right = mid - 1;
        }
    }

    cout << ans << endl;

    return 0;
}
```

Can you find it?

  

Contest Problem

Time Limit：C/C++ 3000MS，Other 6000MS  
Memory Limit：C/C++ 9MB，Other 18MB

Description

Give you three sequences of numbers A, B, C, then we give you a number X. Now you need to calculate if you can find the three numbers Ai, Bj, Ck, which satisfy the formula Ai+Bj+Ck = X.  

Input

There are many cases. Every data case is described as followed: In the first line there are three integers L, N, M, in the second line there are L integers represent the sequence A, in the third line there are N integers represent the sequences B, in the forth line there are M integers represent the sequence C. In the fifth line there is an integer S represents there are S integers X to be calculated. 1<=L, N, M<=500, 1<=S<=1000. all the integers are 32-integers.  

Output

For each case, firstly you have to print the case number as the form “Case d:”, then for the S queries, you calculate if the formula can be satisfied or not. If satisfied, you print “YES”, otherwise print “NO”.  

Sample Input 1 

3 3 3
1 2 3
1 2 3
1 2 3
3
1
4
10

Sample Output 1 

Case 1:
NO
YES
NO

```C++
#include <iostream>
#include <vector>
#include <algorithm>

using namespace std;

void solve(int L, int N, int M, int caseNum) {
    vector<int> A(L), B(N), C(M);
    for (int &x : A) cin >> x;
    for (int &x : B) cin >> x;
    for (int &x : C) cin >> x;

    // 1. 预处理 A+B 的所有组合
    vector<int> sumAB;
    sumAB.reserve(L * N);
    for (int i = 0; i < L; ++i) {
        for (int j = 0; j < N; ++j) {
            sumAB.push_back(A[i] + B[j]);
        }
    }

    // 2. 排序以便二分查找
    sort(sumAB.begin(), sumAB.end());
    // 去重可以进一步优化，但不是必须
    sumAB.erase(unique(sumAB.begin(), sumAB.end()), sumAB.end());

    int S;
    cin >> S;
    cout << "Case " << caseNum << ":" << endl;

    while (S--) {
        int X;
        cin >> X;
        bool found = false;

        // 3. 遍历 C，二分找 X - C[k]
        for (int k = 0; k < M; ++k) {
            int target = X - C[k];
            if (binary_search(sumAB.begin(), sumAB.end(), target)) {
                found = true;
                break;
            }
        }

        if (found) cout << "YES" << endl;
        else cout << "NO" << endl;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int L, N, M, caseNum = 1;
    while (cin >> L >> N >> M) {
        solve(L, N, M, caseNum++);
    }
    return 0;
}
```
