Description

An array of size _n_ ≤ 106 is given to you. There is a sliding window of size _k_ which is moving from the very left of the array to the very right. You can only see the _k_ numbers in the window. Each time the sliding window moves rightwards by one position. Following is an example:  
The array is [1 3 -1 -3 5 3 6 7], and _k_ is 3.

|Window position|Minimum value|Maximum value|
|---|---|---|
|[1  3  -1] -3  5  3  6  7|-1|3|
|1 [3  -1  -3] 5  3  6  7|-3|3|
|1  3 [-1  -3  5] 3  6  7|-3|5|
|1  3  -1 [-3  5  3] 6  7|-3|5|
|1  3  -1  -3 [5  3  6] 7|3|6|
|1  3  -1  -3  5 [3  6  7]|3|7|

Your task is to determine the maximum and minimum values in the sliding window at each position.

Input

The input consists of two lines. The first line contains two integers _n_ and _k_ which are the lengths of the array and the sliding window. There are _n_ integers in the second line.

Output

There are two lines in the output. The first line gives the minimum values in the window at each position, from left to right, respectively. The second line gives the maximum values.

Sample Input 1 

8 3
1 3 -1 -3 5 3 6 7

Sample Output 1 

-1 -3 -3 -3 3 3
3 3 5 5 6 7

```C++
#include <deque>
#include <iostream>
#include <vector>

using namespace std;

void solve()
{
    int n, k;
    if (!(cin >> n >> k))
        return;

    vector<int> a(n);
    for (int i = 0; i < n; ++i)
        cin >> a[i];

    // 单调队列存索引
    deque<int> q;

    // 1. 求最小值
    for (int i = 0; i < n; ++i)
    {
        // 如果队头超出窗口范围，弹出
        while (!q.empty() && q.front() <= i - k)
            q.pop_front();
        // 如果新元素比队尾元素还小，队尾没用了
        while (!q.empty() && a[q.back()] >= a[i])
            q.pop_back();
        q.push_back(i);
        // 窗口形成后开始输出
        if (i >= k - 1)
            cout << a[q.front()] << (i == n - 1 ? "" : " ");
    }
    cout << endl;

    q.clear();

    // 2. 求最大值
    for (int i = 0; i < n; ++i)
    {
        while (!q.empty() && q.front() <= i - k)
            q.pop_front();
        // 如果新元素比队尾元素还大，队尾没用了
        while (!q.empty() && a[q.back()] <= a[i])
            q.pop_back();
        q.push_back(i);
        if (i >= k - 1)
            cout << a[q.front()] << (i == n - 1 ? "" : " ");
    }
    cout << endl;
}

int main()
{
    // 加速 I/O，在大数据量下非常重要
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    solve();
    return 0;
}
```
