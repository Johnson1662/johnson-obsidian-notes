# 位运算
![](assets/第二章%20信息的表示和处理/file-20260206212112153.png)
```C
bool allNibblesGreaterThan9(uint32_t X) {
    // 1. 提取各个权重位
    uint32_t x3 = X & 0x88888888; // 8
    uint32_t x2 = X & 0x44444444; // 4
    uint32_t x1 = X & 0x22222222; // 2
    
    // 2. 运用公式: f = x3 & (x2 | x1)
    // 我们把所有位都对齐到 x1 的位置（即每个 4 位组的第 2 位）进行运算
    uint32_t match = (x3 >> 2) & ((x2 >> 1) | x1);
    
    // 3. 检查结果
    // 如果 8 个十六进制位都符合，那么 match 应该是 0x22222222
    return match == 0x22222222;
}
```
